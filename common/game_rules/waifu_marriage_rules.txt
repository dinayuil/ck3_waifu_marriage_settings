# 控制“召唤婚舰参战”交互（玩家 ↔ 誓约纸片人）
# Controls the "call sworn waifu to war" interaction (player <-> sworn 2D character)
#
# 选项 / Options:
#   enabled      —— 双向：玩家可召唤婚舰，婚舰（AI）也可召唤玩家（默认）

	categories = {

	}

	waifu_call_to_war_player_only = { }    # 仅玩家召唤婚舰 / Player calls waifu only
	waifu_call_to_war_reverse_only = { }   # 仅婚舰召唤玩家 / Waifu calls player only
	waifu_call_to_war_disabled = { }       # 关闭 / Disabled
}
