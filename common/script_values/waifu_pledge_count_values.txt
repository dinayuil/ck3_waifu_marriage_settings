# 覆盖婚舰通用库的 to_*_shiyue_interaction_number 脚本值
# 原版用 every_courtier_or_guest（只统计廷臣/宾客），导致封臣、自为主人的纸片人不被统计。
# 这里在【玩家领地内】统计：玩家的【廷臣/宾客】+【封臣】（every_courtier_or_guest + every_vassal），
# 避免全图遍历造成卡顿，同时包含封臣、自己也是主人的纸片人。
# 配合 common/scripted_triggers/waifu_pledge_expansion_triggers.txt 里的 can_be_oath_by_ACTOR_shown 覆盖。

to_wsg_shiyue_interaction_number = {
	value = 0
	every_courtier_or_guest = {
		limit = {
			warshipgirl_trigger = yes
			can_be_oath_by_ACTOR_shown = { ACTOR = root }
			can_be_oath_by_ACTOR = { ACTOR = root }
		}
		add = 1
	}
	every_vassal = {
		limit = {
			warshipgirl_trigger = yes
			can_be_oath_by_ACTOR_shown = { ACTOR = root }
			can_be_oath_by_ACTOR = { ACTOR = root }
		}
		add = 1
	}
}
