# 纸片人婚姻设置 / Waifu Marriage Settings
# 覆盖原版 marriage_interaction_valid_target_trigger，加入：
#   1. 纸片人只能和纸片人结婚的检查
#   2. 同性纸片人对绕过性别限制
# 并在文件末尾追加两个全新触发器（不覆盖任何现有触发器）。
#
# 兼容性说明：若其他 mod 也覆盖此触发器，需手动合并；

marriage_interaction_valid_target_trigger = {

	# ══════════════════════════════════════════════════════════════════
	# 1. 纸片人婚姻设置：纸片人只能和纸片人结婚
	# Waifu-only restriction: waifus can only marry other waifus
	# ══════════════════════════════════════════════════════════════════
	trigger_if = {
		limit = {
			exists = global_var:anime_core_loaded
			has_game_rule = waifu_marriage_settings_enabled
			OR = {
				scope:secondary_actor     = { waifu_portrait_trigger = yes }
				scope:secondary_recipient = { waifu_portrait_trigger = yes }
			}
		}
		custom_description = {
			text = waifu_must_marry_waifu
			subject = scope:secondary_actor
			object  = scope:secondary_recipient
			AND = {
				scope:secondary_actor     = { waifu_portrait_trigger = yes }
				scope:secondary_recipient = { waifu_portrait_trigger = yes }
			}
		}
	}

	# ══════════════════════════════════════════════════════════════════
	# 以下为原版内容（含同性婚姻绕过修改）
	# Vanilla content below (with same-sex bypass modification)
	# ══════════════════════════════════════════════════════════════════

	# Clergy cannot marry if their faith doesn't allow it
	trigger_if = {
		limit = {
			scope:secondary_actor = {
				is_clergy = yes
				NOT = {
					faith = {
						has_doctrine_parameter = clergy_can_marry
					}
				}
			}
		}
		custom_description = {
			text = character_is_clergy_and_cannot_marry
			subject = scope:secondary_actor
			object = scope:secondary_recipient
			scope:secondary_actor = {
				is_clergy = no
			}
		}
	}
	trigger_else_if = {
		limit = {
			scope:secondary_recipient = {
				is_clergy = yes
				NOT = {
					faith = {
						has_doctrine_parameter = clergy_can_marry
					}
				}
			}
		}
		custom_description = {
			text = character_is_clergy_and_cannot_marry
			subject = scope:secondary_recipient
			object = scope:secondary_actor
			scope:secondary_recipient = {
				is_clergy = no
			}
		}
	}
	#Marriage
	trigger_else_if = {
		limit = {
			scope:secondary_actor = { is_adult = yes }
			scope:secondary_recipient = { is_adult = yes }
		}
		scope:secondary_actor = {
			custom_description = {
				text = "can_marry_check_secondary_actor"
				subject = scope:secondary_actor
				object = scope:secondary_recipient
				# 2. 同性纸片人对：绕过性别检查，使用专用触发器
				# Same-sex waifu pair: bypass gender check using dedicated trigger
				trigger_if = {
					limit = {
						exists = global_var:anime_core_loaded
						has_game_rule = waifu_same_sex_marriage_enabled
						waifu_portrait_trigger = yes
						scope:secondary_recipient = { waifu_portrait_trigger = yes }
						sex_same_as = scope:secondary_recipient
					}
					waifu_same_sex_can_marry_trigger = { CHARACTER = scope:secondary_recipient }
				}
				trigger_else = {
					can_marry_character_trigger = { CHARACTER = scope:secondary_recipient }
				}
			}
		}
	}
	#Betrothal
	trigger_else = {
		scope:secondary_actor = {
			custom_description = {
				text = "can_betroth_check_secondary_actor"
				subject = scope:secondary_actor
				object = scope:secondary_recipient
				# 2. 同性纸片人对：绕过性别检查，使用专用触发器
				trigger_if = {
					limit = {
						exists = global_var:anime_core_loaded
						has_game_rule = waifu_same_sex_marriage_enabled
						waifu_portrait_trigger = yes
						scope:secondary_recipient = { waifu_portrait_trigger = yes }
						sex_same_as = scope:secondary_recipient
					}
					waifu_same_sex_can_marry_trigger = { CHARACTER = scope:secondary_recipient }
				}
				trigger_else = {
					can_marry_character_trigger = { CHARACTER = scope:secondary_recipient }
				}
			}
		}
	}

	# If the actor is ai they will not pick a character already considered for concubinage
	trigger_if = {
		limit = {
			scope:actor = { is_ai = yes }
		}
		NOR = {
			scope:secondary_actor = { has_character_flag = has_been_offered_as_concubine }
			scope:secondary_recipient = { has_character_flag = has_been_offered_as_concubine }
		}
	}
}

# 覆盖 allowed_to_marry_same_sex_trigger，增加纸片人同性婚姻豁免
# Overrides allowed_to_marry_same_sex_trigger to add waifu same-sex exemption
#
# 合并两方逻辑：
#   1. 原版：has_game_rule = accepted_same_sex_marriage + 宗教教义检查
#   2. 本 mod：纸片人 + waifu_same_sex_marriage_enabled
#
# 作用：阻止 AI 因信仰不接受同性婚姻而对同性纸片人夫妻发起离婚（+100分来源）
# Effect: prevents AI from using the faith-based +100 divorce modifier against
#         same-sex waifu couples when the waifu same-sex marriage rule is enabled
# ══════════════════════════════════════════════════════════════════════════════
allowed_to_marry_same_sex_trigger = {
	OR = {
		# ── 原版逻辑 / Vanilla logic ─────────────────────────────────────────
		AND = {
			has_game_rule = accepted_same_sex_marriage
			faith = {
				NOR = {
					has_doctrine_parameter = homosexuality_shunned
					has_doctrine_parameter = homosexuality_illegal
				}
			}
		}

		# ── 纸片人同性婚姻豁免 / Waifu same-sex marriage exemption ──────────
		# 纸片人角色 + 启用纸片人同性婚姻规则 → 视为"允许同性婚姻"
		# Waifu character + waifu same-sex rule enabled → treated as allowed
		AND = {
			exists = global_var:anime_core_loaded
			has_game_rule = waifu_same_sex_marriage_enabled
			waifu_portrait_trigger = yes
		}
	}
}


# ══════════════════════════════════════════════════════════════════════════════
# 以下为全新触发器（不覆盖任何现有触发器，仅供本 mod 内部使用）
# New triggers below — do NOT conflict with any vanilla or other mod triggers
# ══════════════════════════════════════════════════════════════════════════════

# 替代 can_marry_character_trigger，专用于同性纸片人对（跳过性别检查）
# Replaces can_marry_character_trigger for same-sex waifu pairs (no gender check)
waifu_same_sex_can_marry_trigger = {
	save_temporary_scope_as = can_marry_char_check
	OR = {
		AND = {
			is_betrothed = yes
			betrothed = $CHARACTER$
		}
		AND = {
			can_marry_trigger = yes
			$CHARACTER$ = { can_marry_trigger = yes }
		}
	}
	trigger_if = {
		limit = {
			$CHARACTER$ = { is_concubine = yes }
		}
		$CHARACTER$ = { is_concubine_of = scope:can_marry_char_check }
	}
	trigger_if = {
		limit = {
			is_concubine = yes
		}
		is_concubine_of = $CHARACTER$
		$CHARACTER$ = { waifu_same_sex_could_marry_trigger = { CHARACTER = scope:can_marry_char_check } }
	}
	trigger_else = {
		waifu_same_sex_could_marry_trigger = { CHARACTER = $CHARACTER$ }
	}
}

# 替代 could_marry_character_trigger，专用于同性纸片人对
# 跳过性别限制，保留近亲禁婚和近期离婚检查
# Replaces could_marry_character_trigger for same-sex waifu pairs
# Skips gender restriction; keeps consanguinity and recent-divorce checks
waifu_same_sex_could_marry_trigger = {
	save_temporary_scope_as = can_marry_check
	can_marry_common_trigger = yes
	$CHARACTER$ = { can_marry_common_trigger = yes }
	# 性别检查已跳过 / Gender check intentionally omitted

	# 是否近期离婚过此角色 / Recent divorce check
	NOT = {
		has_opinion_modifier = {
			modifier = divorced_me_opinion
			target = $CHARACTER$
		}
	}
	# 近亲禁婚 / Consanguinity check
	trigger_if = {
		limit = { NOT = { is_courtier_of = $CHARACTER$ } }
		faith = {
			faith_allows_marriage_consanguinity_trigger = {
				CHARACTER_1 = scope:can_marry_check
				CHARACTER_2 = $CHARACTER$
			}
		}
	}
	trigger_if = {
		limit = { $CHARACTER$ = { NOT = { is_courtier_of = scope:can_marry_check } } }
		$CHARACTER$.faith = {
			faith_allows_marriage_consanguinity_trigger = {
				CHARACTER_1 = scope:can_marry_check
				CHARACTER_2 = $CHARACTER$
			}
		}
	}
	# 不能和自己结婚 / Cannot marry self
	NOT = {
		scope:can_marry_check = { is_spouse_of = $CHARACTER$ }
	}
}
